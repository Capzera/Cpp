HuffmanNodeData* HuffmanTreeModule::nodeAt(int id) {
    if (id < 0 || id >= static_cast<int>(m_nodes.size())) return nullptr;
    return &m_nodes[id];
}

const HuffmanNodeData* HuffmanTreeModule::nodeAt(int id) const {
    if (id < 0 || id >= static_cast<int>(_nodes.size())) return nullptr;
    return &m_nodes[id];
}

bool HuffmanTreeModule::subtreeHas(int nodeId, int leafId) const {
    const HuffmanNodeData* node = nodeAt(nodeId);
    if (!node) return false;
    if (node->original) return node->id == leafId;
    return subtreeHas(node->left, leafId) || subtreeHas(node->right, leafId);
}

void HuffmanTreeModule::countLeaves(int id, int& leaves) const {
    const HuffmanNodeData* node = nodeAt(id);
    if (!node) return;
    if (node->original) { ++leaves; return; }
    countLeaves(node->left, leaves);
    countLeaves(node->right, leaves);
}

void HuffmanTreeModule::computeStateLayout(HuffmanSimState& state) const {
    state.xs.clear();
    if (state.entries.empty()) return;

    std::map<int, double> memo;
    std::vector<int> roots = componentRoots(state);
    std::sort(roots.begin(), roots.end(), [this](int a, int b) {
        if (m_nodes[a].weight != m_nodes[b].weight) return m_nodes[a].weight < m_nodes[b].weight;
        return a < b;
    });

    std::vector<double> widths;
    double total = 0.0;
    for (int root : roots) {
        const double w = subtreeWidth(root, memo);
        widths.push_back(w);
        total += w;
    }
    const double gap = kTreeGap;
    total += gap * qMax(0, static_cast<int>(roots.size()) - 1);

    double cursor = -total / 2.0;
    for (size_t i = 0; i < roots.size(); ++i) {
        const double w = widths[i];
        layoutComponent(roots[i], cursor + w / 2.0, state.xs, memo);
        cursor += w + gap;
    }

    double minX = 0.0;
    double maxX = 0.0;
    bool first = true;
    for (const auto& kv : state.xs) {
        if (first) { minX = kv.second; maxX = kv.second; first = false; continue; }
        minX = qMin(minX, kv.second);
        maxX = qMax(maxX, kv.second);
    }
    if (first) return;
    const double shift = -((minX + maxX) / 2.0);
    for (auto& kv : state.xs) kv.second += shift;
}

void HuffmanTreeModule::layoutComponentSym(int id, double rootX, std::map<int, double>& xs,
                                           std::map<int, HuffmanExtent>& memo) const {
    const HuffmanNodeData* node = nodeAt(id);
    if (!node) return;
    if (node->original) {
        xs[id] = rootX;
        return;
    }
    const HuffmanExtent le = subtreeExtent(node->left, memo);
    const HuffmanExtent re = subtreeExtent(node->right, memo);
    const double leftExtent = qMax(qAbs(le.left), qAbs(le.right));
    const double rightExtent = qMax(qAbs(re.left), qAbs(re.right));
    const double needed = leftExtent + rightExtent + kTreeGap;
    const double d = qMax(needed / 2.0, kLeafSpan / 2.0);
    const double leftX = rootX - d;
    const double rightX = rootX + d;
    xs[id] = rootX;
    layoutComponentSym(node->left, leftX, xs, memo);
    layoutComponentSym(node->right, rightX, xs, memo);
}

void HuffmanTreeModule::computeStateLayout(HuffmanSimState& state) const {
    state.xs.clear();
    if (state.entries.empty()) return;

    std::map<int, HuffmanExtent> memo;
    std::vector<int> roots = componentRoots(state);
    std::stable_sort(roots.begin(), roots.end(), [this](int a, int b) {
        auto ia = m_compRank.find(a);
        auto ib = m_compRank.find(b);
        const int ra = (ia == m_compRank.end()) ? a : ia->second;
        const int rb = (ib == m_compRank.end()) ? b : ib->second;
        if (ra != rb) return ra < rb;
        return a < b;
    });
    state.roots = roots;

    std::vector<HuffmanExtent> extents;
    double total = 0.0;
    for (int root : roots) {
        const HuffmanExtent e = subtreeExtent(root, memo);
        extents.push_back(e);
        total += (e.right - e.left);
    }
    const double gap = kTreeGap;
    total += gap * qMax(0, static_cast<int>(roots.size()) - 1);

    double cursor = -total / 2.0;
    for (size_t i = 0; i < roots.size(); ++i) {
        const HuffmanExtent& e = extents[i];
        const double w = e.right - e.left;
        layoutComponentSym(roots[i], cursor - e.left, state.xs, memo);
        cursor += w + gap;
    }

    double minX = 0.0;

HuffmanLayoutInfo HuffmanTreeModule::subtreeLayout(int id, std::map<int, HuffmanLayoutInfo>& memo) const {
